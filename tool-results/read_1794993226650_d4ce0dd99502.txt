    46→const CatalogView = ({ categories = [], lockedCategory = null, lockedSub = null }) => {

    48→  const setFilter = useShop((state) => state.setFilter)
    49→  const resetFilters = useShop((state) => state.resetFilters)
    50→  const navigate = useShop((state) => state.navigate)
    51→  const searchParams = useSearchParams()
    52→  const t = useT()
    53→
    54→  const urlSearch = searchParams.get('search') || ''
    55→  const [searchInput, setSearchInput] = useState(urlSearch)
    56→  const [drawerOpen, setDrawerOpen] = useState(false)
    57→  const debouncedSearch = useDebounce(searchInput, 400)
    58→

    62→  if (lastUrlSearch !== urlSearch) {
    63→    setLastUrlSearch(urlSearch)

    65→  }
    66→
    67→  useEffect(() => {
    68→    // Читаем актуальный стор через getState(): депсы только по debouncedSearch,

    70→    if (debouncedSearch !== useShop.getState().filters.search) setFilter({ search: debouncedSearch })
    71→  }, [debouncedSearch, setFilter])
    72→
    73→  const buildUrl = useCallback(
    74→    (offset, pageSize) => {
    75→      const params = new URLSearchParams()

    77→      else if (filters.category && filters.category !== 'all') params.set('category', filters.category)
    78→      if (lockedSub) params.set('sub', lockedSub)
    79→      if (filters.gender && filters.gender !== 'all') params.set('gender', filters.gender)
    80→      if (filters.search) params.set('search', filters.search)
    81→      if (filters.sort && filters.sort !== 'new') params.set('sort', filters.sort)
    82→      if (filters.label && filters.label !== 'all') params.set('label', filters.label)
    83→      if (filters.min > PRICE_LIMITS.min) params.set('min', String(filters.min))
    84→      if (filters.max < PRICE_LIMITS.max) params.set('max', String(filters.max))

    86→      if (filters.discount) params.set('discount', '1')

    88→      params.set('offset', String(offset))
    89→      params.set('limit', String(pageSize))
    90→      return `/api/products?${params.toString()}`
    91→    },
    92→    [filters, lockedCategory, lockedSub]
    93→  )
    94→

    97→    fetcher: API.get,
    98→    onError: (error) => toast.error(t('toast.loadError'), { description: error.message }),
    99→  })
   100→
   101→  const toggleSize = (size) => {
   102→    const next = filters.sizes.includes(size) ? filters.sizes.filter((s) => s !== size) : [...filters.sizes, size]
   103→    setFilter({ sizes: next })
   104→  }
   105→
   106→  const activeCategory = useMemo(
   107→    () => categories.find((category) => category.slug === (lockedCategory || filters.category)),
   108→    [categories, lockedCategory, filters.category]
   109→  )
   110→
   111→  const activeRating = filters.rating || 0
   112→
   113→  // Счётчик активных фильтров (бейдж на плавающей кнопке и в шапке щита)
   114→  const activeCount =
   115→    (filters.sizes?.length || 0) +
   116→    (filters.gender && filters.gender !== 'all' ? 1 : 0) +
   117→    (filters.label && filters.label !== 'all' ? 1 : 0) +
   118→    (filters.discount ? 1 : 0) +
   119→    (activeRating > 0 ? 1 : 0) +
   120→    (filters.min > PRICE_LIMITS.min || filters.max < PRICE_LIMITS.max ? 1 : 0) +
   121→    (searchInput || filters.search ? 1 : 0)
   122→
   123→  // Заголовок страницы: подкатегория (если открыта её SEO-страница) или категория
   124→  const activeSubcategory = lockedSub && activeCategory
   125→    ? (activeCategory.subcategories || []).find((sub) => sub.slug === lockedSub) || null
   126→    : null
   127→  const heading = activeSubcategory
   128→    ? activeSubcategory.name
   129→    : activeCategory
   130→      ? activeCategory.name
   131→      : t('catalog.title')
   132→  const subHeading = activeSubcategory
   133→    ? activeCategory.subtitle
   134→    : activeCategory
   135→      ? activeCategory.subtitle
   136→      : t('catalog.itemsShort')
   137→
   138→  const resetAll = () => {
   139→    resetFilters()
   140→    setSearchInput('')
   141→  }
   142→

   144→  const chipClass = (active, tone = 'primary') =>
   145→    `rounded-full px-3.5 py-2 font-display text-xs font-bold transition-all ${
   146→      active ? (tone === 'orange' ? CHIP_ACTIVE_ORANGE : CHIP_ACTIVE) : CHIP_GHOST
   147→    }`
   148→
   149→  // Единый набор секций фильтров: рендерится в сайдбаре (десктоп) и в боттомщите (мобильные).
   150→  // withSort добавляет секцию сортировки внутрь мобильного щита (на десктопе сортировка живёт в тулбаре над сеткой).

   152→    const groupsWithSubs = categories.filter((category) => category.subcategories?.length)
   153→
   154→    return (
   155→      <div className="space-y-5">
   156→        {withSort ? (
   157→          <section aria-label={t('catalog.sort')}>
   158→            <SectionTitle>{t('catalog.sort')}</SectionTitle>
   159→            <Select value={filters.sort} onValueChange={(value) => setFilter({ sort: value })}>
   160→              <SelectTrigger
   161→                className="h-10 w-full rounded-full border-white/20 bg-white/5 text-sm font-semibold"
   162→                aria-label={t('catalog.sort')}
   163→              >
   164→                <SelectValue placeholder={t('sort.new')} />
   165→              </SelectTrigger>
   166→              <SelectContent className="rounded-2xl border-white/15 bg-[#181335]/95 backdrop-blur-xl">
   167→                {SORT_OPTIONS.map((option) => (
   168→                  <SelectItem key={option.value} value={option.value} className="rounded-xl font-semibold">
   169→                    {t(option.key)}
   170→                  </SelectItem>
   171→                ))}
   172→              </SelectContent>
   173→            </Select>
   174→          </section>
   175→        ) : null}
   176→
   177→        {/* Поиск */}
   178→        <section aria-label={t('catalog.search')}>
   179→          <SectionTitle>{t('catalog.search')}</SectionTitle>
   180→          <Input
   181→            value={searchInput}
   182→            onChange={(event) => setSearchInput(event.target.value)}
   183→            placeholder={t('catalog.searchPlaceholder')}
   184→            className="h-10 rounded-full border-white/20 bg-white/5 text-sm font-semibold"
   185→            aria-label={t('header.search')}
   186→          />
   187→        </section>
   188→
   189→        {/* Категории — ссылки на SEO-страницы, кнопки в столбик как в админке */}
   190→        <section aria-label={t('catalog.categories')}>
   191→          <SectionTitle>{t('catalog.categories')}</SectionTitle>
   192→          <div role="group" aria-label={t('catalog.categories')} className="flex flex-col gap-1.5">
   193→            <button type="button" onClick={() => navigate('catalog')} aria-pressed={!lockedCategory} className={ROW_CLASS(!lockedCategory)}>
   194→              {t('catalog.all')}
   195→            </button>
   196→            {categories.map((category) => (
   197→              <button
   198→                key={category.id}
   199→                type="button"
   200→                onClick={() => navigate('category', { slug: category.slug })}
   201→                aria-pressed={lockedCategory === category.slug}
   202→                className={ROW_CLASS(lockedCategory === category.slug)}
   203→              >
   204→                {category.name}
   205→                <span className="text-xs font-semibold opacity-70">{category.count ?? ''}</span>
   206→              </button>
   207→            ))}

   209→        </section>
   210→
   211→        {/* Подкатегории: столбик кнопок — на залоченной странице одна группа, на /catalog — раскрывающиеся группы по категориям */}
   212→        {lockedCategory ? (
   213→          <section aria-label={t('catalog.subcategories')}>
   214→            <SectionTitle>{t('catalog.subcategories')}</SectionTitle>
   215→            <div role="group" aria-label={t('catalog.subcategories')} className="flex flex-col gap-1.5">
   216→              <button
   217→                type="button"
   218→                onClick={() => navigate('category', { slug: lockedCategory })}
   219→                aria-pressed={!lockedSub}
   220→                className={ROW_CLASS(!lockedSub, 'orange')}
   221→              >
   222→                {t('category.allSubcategories')}
   223→              </button>
   224→              {(activeCategory?.subcategories || []).map((sub) => (
   225→                <button
   226→                  key={sub.id}
   227→                  type="button"
   228→                  onClick={() => navigate('subcategory', { slug: lockedCategory, sub: sub.slug })}
   229→                  aria-pressed={lockedSub === sub.slug}
   230→                  className={ROW_CLASS(lockedSub === sub.slug, 'orange')}
   231→                >
   232→                  {sub.name}
   233→                </button>
   234→              ))}
   235→            </div>
   236→          </section>
   237→        ) : groupsWithSubs.length ? (
   238→          <section aria-label={t('catalog.subcategories')}>
   239→            <details className="group rounded-2xl border border-white/10 bg-white/5">
   240→              <summary className="flex cursor-pointer list-none items-center justify-between gap-2 px-4 py-3 font-display text-[11px] font-bold uppercase tracking-[0.2em] text-purple-200 transition-colors hover:text-white [&::-webkit-details-marker]:hidden">
   241→                {t('catalog.subcategories')}
   242→                <ChevronDown className="h-4 w-4 shrink-0 transition-transform group-open:rotate-180" aria-hidden />
   243→              </summary>
   244→              <div className="scrollbar-glass max-h-72 space-y-4 overflow-y-auto px-3 pb-4 pt-1">
   245→                {groupsWithSubs.map((category) => (
   246→                  <div key={category.id}>
   247→                    <p className="px-1.5 pb-1.5 text-[11px] font-bold uppercase tracking-wider text-purple-300/80">{category.name}</p>
   248→                    <div className="flex flex-col gap-1.5">
   249→                      {category.subcategories.map((sub) => (
   250→                        <button
   251→                          key={sub.id}
   252→                          type="button"
   253→                          onClick={() => navigate('subcategory', { slug: category.slug, sub: sub.slug })}
   254→                          className="glass-btn glass-btn-ghost flex w-full items-center justify-between rounded-2xl px-3.5 py-2.5 text-left font-display text-sm font-bold text-purple-200 transition-all hover:text-white"
   255→                        >
   256→                          {sub.name}
   257→                          <ChevronRight className="h-3.5 w-3.5 shrink-0 text-orange-300/70" aria-hidden />
   258→                        </button>
   259→                      ))}
   260→                    </div>
   261→                  </div>
   262→                ))}
   263→              </div>
   264→            </details>
   265→          </section>
   266→        ) : null}
   267→
   268→        {/* Пол — кнопки в столбик как в админке */}
   269→        <section aria-label={t('catalog.gender')}>
   270→          <SectionTitle>{t('catalog.gender')}</SectionTitle>
   271→          <div role="group" aria-label={t('catalog.gender')} className="flex flex-col gap-1.5">
   272→            {GENDERS.map((gender) => (
   273→              <button
   274→                key={gender.value}
   275→                type="button"
   276→                onClick={() => setFilter({ gender: gender.value })}
   277→                aria-pressed={filters.gender === gender.value}
   278→                className={ROW_CLASS(filters.gender === gender.value)}
   279→              >
   280→                {t(gender.key)}
   281→              </button>
   282→            ))}
   283→          </div>
   284→        </section>
   285→
   286→        {/* Цена: поля ввода + range-слайдер от минимума до максимума */}
   287→        <section aria-label={t('catalog.price')}>
   288→          <SectionTitle>{t('catalog.price')}</SectionTitle>
   289→          <div className="flex items-center gap-2">
   290→            <Input
   291→              type="number"
   292→              min={PRICE_LIMITS.min}
   293→              max={PRICE_LIMITS.max}
   294→              value={filters.min}
   295→              onChange={(event) => {
   296→                const parsed = Number(event.target.value)
   297→                const safe = Number.isFinite(parsed) ? parsed : PRICE_LIMITS.min
   298→                setFilter({ min: Math.min(Math.max(PRICE_LIMITS.min, safe), PRICE_LIMITS.max) })
   299→              }}
   300→              className="h-10 rounded-full border-white/20 bg-white/5 text-sm font-semibold"
   301→              aria-label={t('catalog.priceMin')}

   304→              —
   305→            </span>
   306→            <Input
   307→              type="number"
   308→              min={PRICE_LIMITS.min}
   309→              max={PRICE_LIMITS.max}
   310→              value={filters.max}
   311→              onChange={(event) => {
   312→                const raw = event.target.value
   313→                const parsed = Number(raw)
   314→                const safe = raw === '' || !Number.isFinite(parsed) ? PRICE_LIMITS.max : parsed
   315→                setFilter({ max: Math.min(PRICE_LIMITS.max, Math.max(PRICE_LIMITS.min, safe)) })
   316→              }}
   317→              className="h-10 rounded-full border-white/20 bg-white/5 text-sm font-semibold"
   318→              aria-label={t('catalog.priceMax')}
   319→            />
   320→          </div>
   321→
   322→          {/* Range-слайдер: два ползунка (от/до), синхронизированы с полями ввода */}
   323→          <div className="mt-3 px-1.5">
   324→            <Slider
   325→              value={[filters.min, filters.max]}
   326→              min={PRICE_LIMITS.min}
   327→              max={PRICE_LIMITS.max}
   328→              step={1}
   329→              minStepsBetweenThumbs={1}
   330→              onValueChange={([low, high]) => setFilter({ min: low, max: high })}
   331→              aria-label={t('catalog.price')}
   332→              className="[&_[data-slot=slider-range]]:bg-gradient-to-r [&_[data-slot=slider-range]]:from-purple-500 [&_[data-slot=slider-range]]:to-orange-400 [&_[data-slot=slider-thumb]]:size-4.5 [&_[data-slot=slider-thumb]]:border-2 [&_[data-slot=slider-thumb]]:border-orange-300 [&_[data-slot=slider-thumb]]:bg-white [&_[data-slot=slider-thumb]]:shadow-[0_0_10px_rgba(249,115,22,0.5)] [&_[data-slot=slider-track]]:h-2 [&_[data-slot=slider-track]]:border [&_[data-slot=slider-track]]:border-white/15 [&_[data-slot=slider-track]]:bg-white/10"
   333→            />
   334→            <div className="mt-1.5 flex justify-between text-[10px] font-bold tracking-wider text-purple-300/70">
   335→              <span>€{PRICE_LIMITS.min}</span>
   336→              <span>€{PRICE_LIMITS.max}</span>
   337→            </div>
   338→          </div>
   339→        </section>
   340→
   341→        {/* Размеры: одежда + обувь — компактные чипы лентой, как было */}
   342→        <section aria-label={t('catalog.sizes')}>
   343→          <SectionTitle>{t('catalog.sizes')}</SectionTitle>
   344→          <div role="group" aria-label={t('catalog.sizes')} className="space-y-3">
   345→            <div className="flex flex-wrap gap-2">
   346→              {CLOTHING_SIZES.map((size) => (
   347→                <button
   348→                  key={size}
   349→                  type="button"
   350→                  onClick={() => toggleSize(size)}
   351→                  aria-pressed={filters.sizes.includes(size)}
   352→                  className={chipClass(filters.sizes.includes(size), 'orange')}
   353→                >
   354→                  {size}
   355→                </button>
   356→              ))}
   357→            </div>
   358→            <div className="flex flex-wrap items-center gap-2">
   359→              <span className="text-[10px] font-bold uppercase tracking-wider text-purple-300/70">
   360→                {t('catalog.sizeShoes')}
   361→              </span>
   362→              {SHOE_SIZES.map((size) => (
   363→                <button
   364→                  key={size}
   365→                  type="button"
   366→                  onClick={() => toggleSize(size)}
   367→                  aria-pressed={filters.sizes.includes(size)}
   368→                  className={chipClass(filters.sizes.includes(size), 'orange')}
   369→                >
   370→                  {size}
   371→                </button>
   372→              ))}
   373→            </div>
   374→          </div>
   375→        </section>
   376→
   377→        {/* Бейджи NEW / TOP / SALE — компактные чипы лентой, как было (повторный клик снимает выбор) */}
   378→        <section aria-label={t('catalog.badges')}>
   379→          <SectionTitle>{t('catalog.badges')}</SectionTitle>
   380→          <div role="group" aria-label={t('catalog.badges')} className="flex flex-wrap gap-2">
   381→            {LABEL_FILTERS.map((label) => (
   382→              <button
   383→                key={label}
   384→                type="button"
   385→                onClick={() => setFilter({ label: filters.label === label ? 'all' : label })}
   386→                aria-pressed={filters.label === label}
   387→                className={chipClass(filters.label === label, 'orange')}
   388→              >
   389→                {label.toUpperCase()}
   390→              </button>
   391→            ))}
   392→          </div>
   393→        </section>
   394→
   395→        {/* Только со скидкой — кнопка-переключатель во всю ширину, как в админке */}
   396→        <section aria-label={t('catalog.discountOnly')}>
   397→          <button
   398→            type="button"
   399→            role="switch"
   400→            aria-checked={filters.discount}
   401→            onClick={() => setFilter({ discount: !filters.discount })}
   402→            className={ROW_CLASS(filters.discount, 'orange')}
   403→          >
   404→            <span>{t('catalog.discountOnly')}</span>
   405→            <span
   406→              aria-hidden
   407→              className={`h-5 w-9 rounded-full border transition-colors ${filters.discount ? 'border-orange-300/70 bg-orange-400/80' : 'border-white/25 bg-white/10'}`}
   408→            >
   409→              <span
   410→                className={`block h-4 w-4 translate-y-[1px] rounded-full bg-white shadow transition-transform ${filters.discount ? 'translate-x-[17px]' : 'translate-x-[2px]'}`}
   411→              />
   412→            </span>
   413→          </button>
   414→        </section>
   415→
   416→        {/* Минимальный рейтинг — кнопки в столбик */}
   417→        <section aria-label={t('catalog.rating')}>
   418→          <SectionTitle>{t('catalog.rating')}</SectionTitle>
   419→          <div role="group" aria-label={t('catalog.rating')} className="flex flex-col gap-1.5">
   420→            {RATING_OPTIONS.map((value) => (
   421→              <button
   422→                key={value}
   423→                type="button"
   424→                onClick={() => setFilter({ rating: value })}
   425→                aria-pressed={activeRating === value}
   426→                className={ROW_CLASS(activeRating === value, 'orange')}

   428→                {value === 0 ? t('catalog.ratingAny') : `${value}+`}
   429→              </button>
   430→            ))}
   431→          </div>
   432→        </section>
   433→      </div>
   434→    )
   435→  }
   436→
   437→  return (

   559→          <div className="scrollbar-glass min-h-0 flex-1 overflow-y-auto px-4 py-5">{renderFilters({ withSort: true })}</div>

   583→    </div>
